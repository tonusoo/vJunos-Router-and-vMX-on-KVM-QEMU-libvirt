martin@deb-lab-svr:~$ # Install fresh Debian 13 on a backing disk file. 100 GiB disk in qcow2
martin@deb-lab-svr:~$ # format (sparse file) is created automatically and can be inspected with
martin@deb-lab-svr:~$ # sudo qemu-img info --force-share ~/iasb-class/debian/deb13.qcow2
martin@deb-lab-svr:~$
martin@deb-lab-svr:~$ # OS name/short-id for virt-install "--osinfo" can be seen
martin@deb-lab-svr:~$ # in the output of "osinfo-query os vendor='Debian Project'"
martin@deb-lab-svr:~$ # or "virt-install --osinfo list"
martin@deb-lab-svr:~$ sudo virt-install --osinfo debian13 \
    --name a-svr1 \
    --events on_crash=restart \
    --memory 16384 \
    --vcpus 2 \
    --location https://deb.debian.org/debian/dists/trixie/main/installer-amd64/ \
    --disk ~/iasb-class/debian/deb13.qcow2,size=100 \
    --network bridge="a-br-ext,target=a-svr1-vnet0" \
    --graphics none \
    --extra-args "console=ttyS0"
