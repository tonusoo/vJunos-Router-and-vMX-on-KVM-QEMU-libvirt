[Network]
# Allows systemd-networkd to configure a link even if it has no carrier.
# The bridge interface with no members will have a NO-CARRIER link status
# and without ConfigureWithoutCarrier=yes the bridge interface will be stuck
# in "configuring" state and this in turn means that the systemd-networkd-wait-online
# service will timeout.
